    \section{VM Images}
    \begin{frame}
        \frametitle{Setup}
        \begin{itemize}
            \item three VM Images where backed up (like in the paper) with a fixed chunk size of one MiB
            \item they were pruned beginning with the oldest manifest
            \item the time required for the pruning commands to complete was recorded
            \item in all four programs deletion happens in two phases:
            \begin{enumerate}
                \item deletion of the manifest (borg delete or fossil collection, blue bar)
                \item freeing of now unused disk space (borg compact or fossil deletion, red bar)
            \end{enumerate}
        \end{itemize}
    \end{frame}

    \begin{frame}
        \frametitle{Results}
        \begin{center}

    \section{Chunk scaling}
    \begin{frame}
        \frametitle{Setup}
        \begin{itemize}
            \item created a specific number of chunks divided evenly between four manifests
            \item the first manifest was pruned
            \item the time required for the pruning commands to complete was recorded
            \item repositories where located on an ext4 hard drive and caches on a tmpfs
        \end{itemize}
    \end{frame}

    \foreach \f in {64,512,4096,8192,16384,32768,65536,131072,262144,524288}{
        \begin{frame}
            \frametitle{Results (chunk size = \f{} bytes)}
            \begin{center}
                \begin{tikzpicture}
                    \pgfplotsset{
                        table/col sep=comma,
                        height=.85\textheight
                    }
                    \begin{axis}[
                        xlabel=Number of chunks,
                        ylabel=Runtime in seconds,
                        width=0.75\textwidth,
                        legend pos=outer north east,
                        scaled x ticks=base 10:-3
                    ]
                        \addplot table [x=chunks, y expr=\thisrow{Borg 1 deletion}+\thisrow{Borg 1 compaction}] {benchmark/results/chunks\f.csv};
                        \addlegendentry{Borg 1}
                        \addplot table [x=chunks, y expr=\thisrow{Borg 2 deletion}+\thisrow{Borg 2 compaction}] {benchmark/results/chunks\f.csv};
                        \addlegendentry{Borg 2}
                        \addplot table [x=chunks, y expr=\thisrow{Duplicacy collection}+\thisrow{Duplicacy deletion}] {benchmark/results/chunks\f.csv};
                        \addlegendentry{Duplicacy}
